```lean
/-! S036 / EC-C024: raw counter only; no game-admissibility or adjudication. -/
namespace EndgameChess.Rules.RawClock

/-- Source-pawn event, before any promotion. -/
def Pawn (q : Position) (m : Move) : Prop :=
  q.board m.src = some ⟨q.turn, .pawn⟩

/-- Meaning as pawn/capture is asserted only for legal moves. -/
def zeroing (q : Position) (m : Move) : Bool :=
  (q.board m.src == some ⟨q.turn, .pawn⟩) || (q.board m.dst).isSome

/-- Ordinary enemy capture or the kernel en-passant capture event. -/
def CaptureEvent (q : Position) (m : Move) : Prop :=
  (∃ pc, q.board m.dst = some pc ∧ pc.colour ≠ q.turn) ∨
  (Pawn q m ∧ q.ep = some m.dst ∧ m.src.1 ≠ m.dst.1 ∧
    q.board m.dst = none ∧ q.board (m.dst.1, m.src.2) = some ⟨q.turn.opp, .pawn⟩)

/-- Board plies since the last reset, derived from full stored data. -/
def clock {q : Position} (t : RawTrace.Trace q) : Nat :=
  match t with
  | .start => 0
  | @RawTrace.Trace.snoc q prior m _ =>
      if zeroing q m then 0 else clock prior + 1

end EndgameChess.Rules.RawClock
```

```lean
import EndgameChess.Rules.RawClock.Definitions
namespace EndgameChess.Rules.RawClock

/-- Includes every pawn promotion/ep move and every quiet nonpawn move. -/
theorem event_cases (q : Position) (m : Move) :
    (Pawn q m → zeroing q m = true) ∧
    ((∃ pc, q.board m.dst = some pc) → zeroing q m = true) ∧
    (¬ Pawn q m → q.board m.dst = none → zeroing q m = false) := by
  constructor
  · intro h
    change q.board m.src = some ⟨q.turn, .pawn⟩ at h
    simp [zeroing, h]
  constructor
  · rintro ⟨pc, h⟩
    simp [zeroing, h]
  · intro hp hd
    change q.board m.src ≠ some ⟨q.turn, .pawn⟩ at hp
    simp [zeroing, hp, hd]

end EndgameChess.Rules.RawClock
```

```lean
import EndgameChess.Rules.RawClock.Definitions
namespace EndgameChess.Rules.RawClock

/-- Kernel capture-event correspondence; no material-count theorem assumed. -/
theorem legal_zero_iff (q : Position) (m : Move) (hm : isLegal q m = true) :
    zeroing q m = true ↔ Pawn q m ∨ CaptureEvent q m := by
  have hp : isPseudo q m = true := by
    simp only [isLegal, Bool.and_eq_true] at hm
    exact hm.1
  have ht := targetOK_of_isPseudo hp
  by_cases hw : Pawn q m
  · have hw' := hw
    change q.board m.src = some ⟨q.turn, .pawn⟩ at hw'
    simp [zeroing, hw, hw']
  change q.board m.src ≠ some ⟨q.turn, .pawn⟩ at hw
  cases hd : q.board m.dst with
  | none => simp [zeroing, Pawn, CaptureEvent, hd, hw]
  | some pc =>
    have he := (targetOK_some ht hd).1
    have hc : CaptureEvent q m := Or.inl ⟨pc, hd, he⟩
    simp [zeroing, hd, hc]

end EndgameChess.Rules.RawClock
```

```lean
import EndgameChess.Rules.RawClock.Definitions
namespace EndgameChess.Rules.RawClock

/-- Structural induction on full stored history; units are board plies. -/
theorem clock_le_length {q : Position} (t : RawTrace.Trace q) :
    clock t ≤ t.moves.length := by
  induction t with
  | start => exact Nat.le_refl 0
  | @snoc q prior m hm ih =>
    simp only [clock, RawTrace.Trace.moves, List.length_append, List.length_singleton]
    split
    · exact Nat.zero_le _
    · exact Nat.add_le_add_right ih 1

end EndgameChess.Rules.RawClock
```
